// editflow-mono-create の firestore.rules に足すブロック（service cloud.firestore > match /databases/{database}/documents の中）。
// 既存コレクションのルールは変更しない。editflow-team/firestore.rules にも同じ内容が入っていること(次回デプロイで消えないように)。

    // ===== 共同営業台帳 (sales-board, 2026-11-02〜) =====
    // 既存コレクションのルールは一切変更しない(このブロックの追加のみ)。
    // sales_candidates: 1候補=1ドキュメント(ID=@なし小文字ハンドル。__x__ 形式のみ先頭に ~)。
    //   読み取り=誰でも(ミューズ(こまめ)の送信ジョブが REST+APIキーで二重送信チェックするため)。書き込み=owner() のみ。削除不可。
    // それ以外(サムネイル・募集・画像・設定)は読み書きとも owner() のみ。
    function salesTsOk(d, k) {
      return !(k in d) || d[k] == null || d[k] is timestamp;
    }
    function salesCandidateValid(candidateId) {
      let d = request.resource.data;
      return d.keys().hasOnly(['handle','status','genre','followers','source','variant','sentAt','sentFrom','sentBy','note','updatedAt','name','order','approved','rank','text','reel','line','link','hasPhoto','approvedAt','excludedAt','skippedAt','statusAt'])
        && d.keys().hasAll(['handle','status','updatedAt'])
        && d.handle is string
        // ID は @なし小文字ハンドル。__x__ 形式のハンドルだけ Firestore の予約IDなので先頭に ~ を付ける
        && (d.handle.lower().replace('@', '') == candidateId || ('~' + d.handle.lower().replace('@', '')) == candidateId)
        && d.status in ['waiting','sent','replied','meeting','converted','excluded','skipped']
        && (!('followers' in d) || d.followers == null || d.followers is number)
        && (!('order' in d) || d.order == null || d.order is number)
        && (!('approved' in d) || d.approved is bool)
        && (!('hasPhoto' in d) || d.hasPhoto is bool)
        && salesTsOk(d, 'sentAt') && salesTsOk(d, 'approvedAt') && salesTsOk(d, 'excludedAt') && salesTsOk(d, 'skippedAt') && salesTsOk(d, 'statusAt')
        && (!('note' in d) || d.note == null || (d.note is string && d.note.size() <= 2000))
        && (!('text' in d) || d.text == null || (d.text is string && d.text.size() <= 5000))
        && (!('reel' in d) || d.reel == null || (d.reel is string && d.reel.size() <= 3000))
        && (!('line' in d) || d.line == null || (d.line is string && d.line.size() <= 1000))
        && (!('link' in d) || d.link == null || (d.link is string && d.link.size() <= 1000));
    }
    match /sales_candidates/{candidateId} {
      allow get, list: if true;
      allow create, update: if owner() && salesCandidateValid(candidateId);
      allow delete: if false;
    }
    match /sales_thumb_candidates/{candidateId} {
      allow get, list: if owner();
      allow create, update: if owner() && salesCandidateValid(candidateId);
      allow delete: if false;
    }
    match /sales_recruit/{recruitId} {
      allow get, list: if owner();
      allow create, update: if owner()
        && request.resource.data.keys().hasOnly(['kind','category','format','title','poster','price','deadline','summary','url','profileImage','order','status','decidedAt','updatedAt'])
        && request.resource.data.keys().hasAll(['status','updatedAt'])
        && request.resource.data.status in ['new','adopt','reject']
        && (!('order' in request.resource.data) || request.resource.data.order == null || request.resource.data.order is number)
        && salesTsOk(request.resource.data, 'decidedAt');
      allow delete: if false;
    }
    match /sales_photos/{photoId} {
      allow get: if owner();
      allow list: if false;
      allow create, update: if owner()
        && request.resource.data.keys().hasOnly(['data','type','updatedAt'])
        && request.resource.data.data is string && request.resource.data.data.size() < 900000
        && request.resource.data.type in ['image/jpeg','image/png','image/webp'];
      allow delete: if false;
    }
    match /sales_meta/{metaId} {
      allow get, list: if owner();
      allow create, update: if owner()
        && request.resource.data.keys().hasOnly(['title','deadline','target','updatedAt']);
      allow delete: if false;
    }
