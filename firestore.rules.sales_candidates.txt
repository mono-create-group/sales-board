// editflow-mono-create の firestore.rules に足すブロック（service cloud.firestore > match /databases/{database}/documents の中）。
// 既存コレクションのルールは変更しない。editflow-team/firestore.rules にも同じ内容が入っていること(次回デプロイで消えないように)。

    // 共同営業台帳 (sales-board, 2026-11-02)。営業候補は 1候補=1ドキュメント(ID=@なし小文字ハンドル)。
    // 読み取り: ミューズ(こまめ)の送信ジョブが REST+APIキーで二重送信チェックするため未認証でも可。
    // 書き込み: owner() のみ。削除は不可(状態は excluded にする)。許可するフィールドと状態値を固定する。
    // 既存コレクションのルールは一切変更しない(このブロックの追加のみ)。
    function salesCandidateValid(candidateId) {
      let d = request.resource.data;
      return d.keys().hasOnly(['handle','status','genre','followers','source','variant','sentAt','sentFrom','sentBy','note','updatedAt','name','order','approved','rank'])
        && d.keys().hasAll(['handle','status','updatedAt'])
        && d.handle is string
        // ID は @なし小文字ハンドル。__x__ 形式のハンドルだけ Firestore の予約IDなので先頭に ~ を付ける
        && (d.handle.lower().replace('@', '') == candidateId || ('~' + d.handle.lower().replace('@', '')) == candidateId)
        && d.status in ['waiting','sent','replied','meeting','converted','excluded']
        && (!('followers' in d) || d.followers == null || d.followers is number)
        && (!('order' in d) || d.order == null || d.order is number)
        && (!('approved' in d) || d.approved is bool)
        && (!('note' in d) || d.note == null || (d.note is string && d.note.size() <= 2000));
    }

    match /sales_candidates/{candidateId} {
      allow get, list: if true;
      allow create, update: if owner() && salesCandidateValid(candidateId);
      allow delete: if false;
    }
